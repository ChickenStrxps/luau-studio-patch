cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(LuauStudioPatch VERSION 0.5.2 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows x64 and the MSVC toolchain are required")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Select the x64 build target")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/prepare_resources.py"
  --output "${CMAKE_BINARY_DIR}/resources" RESULT_VARIABLE profile_result)
if(NOT profile_result EQUAL 0)
  message(FATAL_ERROR "Patch profile validation failed")
endif()
add_executable(LuauStudioPatch WIN32 src/main.cpp src/studio_installer.cpp "${CMAKE_BINARY_DIR}/resources/profiles.rc")
target_compile_features(LuauStudioPatch PRIVATE cxx_std_17)
target_compile_definitions(LuauStudioPatch PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LuauStudioPatch PRIVATE /W4 /Brepro)
target_link_options(LuauStudioPatch PRIVATE /Brepro /DEBUG:NONE)
target_link_libraries(LuauStudioPatch PRIVATE shell32 bcrypt advapi32 version user32 cabinet comctl32)
